% Slide 7: A zero-coupon bond

\begin{frame}{A zero-coupon bond}
\[p_t(T)=\frac{F}{[1+i_t(T)]^T}.\]\[F=100,\quad T=2,\quad i_t(2)=5\%\quad\Rightarrow\quad p_t(2)=90.70.\]
\end{frame}

% Slide 8: A discount factor prices one future unit

\begin{frame}{A discount factor prices one future unit}
\[d_t(T)=\frac{p_t^{\rm zero}(T)}F=\frac1{[1+i_t(T)]^T}.\]\[p_t=\sum_{k=1}^{n}d_t(k)\,CF_k.\]\eqnote{$CF_k$: the cash payment at date k. Match dates, currency and credit risk.}
\end{frame}

% Slide 9: Replication gives the pricing intuition

\begin{frame}{Replication gives the pricing intuition}
\point{Buy the same cash flows using zero-coupon bonds.}\point{A matching payoff schedule has a matching price.}\eqnote{Same currency, payment dates and credit quality. Borrowing, shorting and trading must be feasible.}
\end{frame}

% Slide 10: A coupon bond is a basket of zeros

\begin{frame}{A coupon bond is a basket of zeros}
\begin{center}\begin{tabular}{@{}lll@{}}\toprule \textbf{Date} & \textbf{Cash flow} & \textbf{Replication}\\\midrule 1 & $4$ & $4$ units of a date-1 zero\\[2mm]
2 & $4$ & $4$ units of a date-2 zero\\[2mm]
3 & $104$ & $104$ units of a date-3 zero\\\bottomrule\end{tabular}\end{center}\[p_t=4d_t(1)+4d_t(2)+104d_t(3).\]
\end{frame}

% Slide 11: Worked price with a flat curve

\begin{frame}{Worked price with a flat curve}
\[p_t=\frac4{1.05}+\frac4{1.05^2}+\frac{104}{1.05^3}=97.28.\]\point{A 4\% coupon trades below face value when the required yield is 5\%.}\eqnote{Annual coupons. Flat 5\% spot curve. Valuation on a coupon date.}
\end{frame}

% Slide 12: Market price determines yield to maturity

\begin{frame}{Market price determines yield to maturity}
\[p_t=\sum_{k=1}^{n}\frac{CF_k}{(1+i^{\rm YTM})^k}.\]\point{Given the price and promised payments, solve for the single rate.}\eqnote{YTM summarizes the bond price. Each dated spot rate can still be different.}
\end{frame}

% Slide 13: Coupon, yield and return

\begin{frame}{Coupon, yield and return}
\begin{center}\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.32\textwidth}>{\raggedright\arraybackslash}p{0.61\textwidth}@{}}\toprule \textbf{Object} & \textbf{What does it measure?}\\\midrule Coupon rate c & Contractual annual payment per unit face\\[2mm]
Yield $i^{\rm YTM}$ & Discount rate matching today’s price\\[2mm]
Holding return H & Cash received plus price change, over your horizon\\\bottomrule\end{tabular}\end{center}\[H_{t,t+1}=\frac{C_{t+1}+p_{t+1}-p_t}{p_t}.\]
\end{frame}

% Slide 14: A 5% yield can produce a one-year loss

\begin{frame}{A 5\% yield can produce a one-year loss}
\begin{center}\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.77\textwidth}>{\raggedright\arraybackslash}p{0.15\textwidth}@{}}\toprule \textbf{Ten-year zero, face 100} & \textbf{Price}\\\midrule Buy at 5\% today & 61.39\\[2mm]
Sell one year later, remaining nine-year yield 6\% & 59.19\\\bottomrule\end{tabular}\end{center}\[H=\frac{59.1898-61.3913}{61.3913}=-3.59\%.\]
\end{frame}

% Slide 15: What holding to maturity locks in

\begin{frame}{What holding to maturity locks in}
\point{A default-free zero locks its nominal payment and annualized return.}\point{Coupon-bond wealth also depends on coupon reinvestment.}\eqnote{Selling early introduces price risk. Inflation and foreign-currency returns add other risks.}
\end{frame}
